import { readFileSync } from 'node:fs';
import { resolve } from 'node:path';
import { pathToFileURL } from 'node:url';
import { describe, expect, it } from 'vitest';

const script = pathToFileURL(resolve('scripts/font-css.mjs')).href;
const { compactFontCss } = (await import(script)) as { compactFontCss(css: string, display?: string): string };
const original = (family: string) =>
  readFileSync(resolve('node_modules/@fontsource-variable', family, 'index.css'), 'utf8')
    .replaceAll('./files/', './')
    .replaceAll(' SC Variable', ' SC');

describe('font display experiments cannot silently change the design default or coverage', () => {
  it.each(['noto-sans-sc', 'noto-serif-sc'])('changes only the display descriptor for every %s face', (family) => {
    const input = original(family),
      normal = compactFontCss(input),
      optional = compactFontCss(input, 'optional');
    expect(normal).toBe(compactFontCss(input, 'swap'));
    expect(normal.match(/font-display:swap/g)).toHaveLength(101);
    expect(optional.match(/font-display:optional/g)).toHaveLength(101);
    expect(optional.replaceAll('font-display:optional', 'font-display:swap')).toBe(normal);
  });
  it.each(['auto', 'block', 'swap;src:url(https://invalid.test)', ''])(
    'rejects unreviewed display value %s',
    (display) => {
      expect(() => compactFontCss(original('noto-sans-sc'), display)).toThrow('Unexpected Fontsource CSS');
    },
  );
  it('refuses changed source display grammar rather than blessing a package drift', () => {
    expect(() =>
      compactFontCss(original('noto-sans-sc').replaceAll('font-display: swap', 'font-display: block'), 'optional'),
    ).toThrow();
  });
});
