def image_paths() -> dict:
    api = ctypes.WinDLL("kernel32", use_last_error=True)
    api.QueryFullProcessImageNameW.argtypes = (
        w.HANDLE,
        w.DWORD,
        w.LPWSTR,
        ctypes.POINTER(w.DWORD),
    )
    api.QueryFullProcessImageNameW.restype = w.BOOL
    api.GetModuleHandleW.argtypes = (w.LPCWSTR,)
    api.GetModuleHandleW.restype = w.HMODULE
    api.GetModuleFileNameW.argtypes = (w.HMODULE, w.LPWSTR, w.DWORD)
    api.GetModuleFileNameW.restype = w.DWORD
    buffer = ctypes.create_unicode_buffer(32768)
    size = w.DWORD(len(buffer))
    if not api.QueryFullProcessImageNameW(
        ctypes.c_void_p(-1), 0, buffer, ctypes.byref(size)
    ):
        raise RuntimeError("identity_process_image_query_failed")
    result = {"process_image": buffer.value, "python_dlls": {}}
    for name in ("python311.dll", "python3.dll"):
        handle = api.GetModuleHandleW(name)
        if not handle:
            result["python_dlls"][name] = None
            continue
        length = api.GetModuleFileNameW(handle, buffer, len(buffer))
        if not length or length >= len(buffer):
            raise RuntimeError("identity_python_dll_query_failed")
        path = Path(buffer.value)
        result["python_dlls"][name] = {
            "path": str(path),
            "sha256": hashlib.sha256(path.read_bytes()).hexdigest(),
        }
    return result


def output_path() -> Path:
    if len(sys.argv) != 2 or len(sys.orig_argv) != 8:
        raise RuntimeError("identity_fixed_arguments_required")
    path = Path(sys.argv[1])
    actual = path.resolve()
    if (
        path.absolute() != actual
        or not actual.is_relative_to(ROOT / "private/tmp")
        or actual.suffix != ".json"
        or actual.exists()
    ):
        raise RuntimeError("identity_new_private_output_required")
    if (
        sys.orig_argv[1:5] != ["-I", "-S", "-B", "-X"]
        or Path(sys.orig_argv[6]).resolve() != Path(__file__).resolve()
    ):
        raise RuntimeError("identity_fixed_flags_required")
    if (
        not sys.flags.isolated
        or not sys.flags.no_site
        or not sys.dont_write_bytecode
        or sys.flags.optimize != 0
    ):
        raise RuntimeError("identity_isolated_unoptimized_required")
    if set(sys._xoptions) != {"pycache_prefix"} or not sys.pycache_prefix:
        raise RuntimeError("identity_only_private_cache_option_required")
    cache = Path(sys.pycache_prefix)
    if (
        cache.absolute() != cache.resolve()
        or not cache.resolve().is_relative_to(ROOT / "private/tmp")
        or not cache.is_dir()
        or any(cache.iterdir())
    ):
        raise RuntimeError("identity_fresh_cache_required")
    return actual
