PRIVATE = ROOT / "private/p8-validation"
OUT = ROOT / "docs/v4/evidence/P8"
COMMIT = sys.argv[1]
assert ROOT.name == "zhiming-v4-p8" and re.fullmatch(r"[a-f0-9]{40}", COMMIT)


def digest(raw):
    return hashlib.sha256(raw).hexdigest()


def save(name, value):
    with (OUT / name).open("x", encoding="utf8", newline="\n") as stream:
        json.dump(value, stream, ensure_ascii=False, indent=2, allow_nan=False)
        stream.write("\n")


def copy(source, name):
    with (OUT / name).open("xb") as stream:
        stream.write(source.read_bytes())


def canonical(value):
    return json.dumps(value, sort_keys=True, ensure_ascii=False, separators=(",", ":")).encode("utf8")

result = subprocess.run(["git", "cat-file", "--batch"], input=batch, capture_output=True, check=True, timeout=30)
cursor, rows = 0, []
for name in paths:
    end = result.stdout.index(b"\n", cursor)
    header = result.stdout[cursor:end].decode("ascii").split()
    assert len(header) == 3 and header[1] == "blob"
    size = int(header[2])
    body = result.stdout[end + 1 : end + 1 + size]
    assert len(body) == size and result.stdout[end + 1 + size] == 10
    cursor = end + 2 + size
    current = (ROOT / name).read_bytes().replace(b"\r\n", b"\n")
    assert digest(body.replace(b"\r\n", b"\n")) == digest(current) == expected[name]
    rows.append({"path": name, "git_blob": header[0], "lf_sha256": expected[name]})
assert cursor == len(result.stdout)
assert digest((PRIVATE / "guarded_environment_validation.py").read_bytes().replace(b"\r\n", b"\n")) == records["full"]["validation_harness_lf_sha256"]
