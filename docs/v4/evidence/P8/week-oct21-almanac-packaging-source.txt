"""Archive bounded local results; no new calculation or model request."""
import copy
import hashlib
import json
import shutil
import xml.etree.ElementTree as ET
from datetime import datetime, timedelta, timezone
from pathlib import Path

ROOT = Path.cwd()
PRIVATE = ROOT / "private/p8-validation"
OUT = ROOT / "docs/v4/evidence/P8"
PREFIX = "week-oct21-almanac"
BEIJING = timezone(timedelta(hours=8))

def read(path):
    return json.loads(path.read_bytes())

def sha(path):
    return hashlib.sha256(path.read_bytes().replace(b"\r\n", b"\n")).hexdigest()

def save(name, value):
    with (OUT / name).open("x", encoding="utf8", newline="\n") as stream:
        stream.write(json.dumps(value, ensure_ascii=False, indent=2) + "\n")

def archive(source, name):
    target = OUT / name
    assert not target.exists()
    shutil.copyfile(source, target)
    return {"path": target.relative_to(ROOT).as_posix(), "lf_sha256": sha(target)}

final_path = PRIVATE / "milestone-week-oct21-almanac-citations-final.start.json"
final = read(final_path)
assert final["exit_code"] == 0 and final["sources_unchanged"] is True
assert len(final["source_snapshot_before"]) == 387
assert final["process_budget"]["configured_limit"] == 4

for tree in (plain_old, plain_new):
